    /* ==========================================================================
       TACTILE PROCEDURAL ASMR SOUND ENGINE (Zero MP3s, Pure Web Audio)
       ========================================================================== */
    class ASMRSoundEngine {
      constructor() {
        this.ctx = null;
        this.muted = false;
      }

      init() {
        if (this.ctx) return;
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        this.ctx = new AudioCtx();
      }

      resume() {
        if (this.ctx && this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
      }

      // Juicy jelly & pancake squish (Pitched sine with rapid envelope sweep)
      playSquish(pitch = 320) {
        if (this.muted || !this.ctx) return;
        this.resume();

        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();

        osc.type = 'sine';
        osc.frequency.setValueAtTime(pitch, now);
        osc.frequency.exponentialRampToValueAtTime(pitch * 0.45, now + 0.12);

        gain.gain.setValueAtTime(0.28, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.13);

        osc.connect(gain);
        gain.connect(this.ctx.destination);

        osc.start(now);
        osc.stop(now + 0.14);
      }

      // Crisp saw slice & layer chop (White noise burst filtered)
      playSlice() {
        if (this.muted || !this.ctx) return;
        this.resume();

        const now = this.ctx.currentTime;
        const bufferSize = this.ctx.sampleRate * 0.08;
        const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const data = buffer.getChannelData(0);

        for (let i = 0; i < bufferSize; i++) {
          data[i] = Math.random() * 2 - 1;
        }

        const noise = this.ctx.createBufferSource();
        noise.buffer = buffer;

        const filter = this.ctx.createBiquadFilter();
        filter.type = 'bandpass';
        filter.frequency.setValueAtTime(2200, now);
        filter.Q.setValueAtTime(3.0, now);

        const gain = this.ctx.createGain();
        gain.gain.setValueAtTime(0.35, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.08);

        noise.connect(filter);
        filter.connect(gain);
        gain.connect(this.ctx.destination);

        noise.start(now);
      }

      // Cheerful chime multiplier ping
      playChime(freq = 587.33) {
        if (this.muted || !this.ctx) return;
        this.resume();

        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();

        osc.type = 'triangle';
        osc.frequency.setValueAtTime(freq, now);

        gain.gain.setValueAtTime(0.22, now);
        gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.35);

        osc.connect(gain);
        gain.connect(this.ctx.destination);

        osc.start(now);
        osc.stop(now + 0.36);
      }

      // Giant Mascot Monch/Munch sound
      playMunch() {
        if (this.muted || !this.ctx) return;
        this.resume();

        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();

        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(140, now);
        osc.frequency.exponentialRampToValueAtTime(45, now + 0.16);

        gain.gain.setValueAtTime(0.4, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.18);

        osc.connect(gain);
        gain.connect(this.ctx.destination);

        osc.start(now);
        osc.stop(now + 0.2);
      }
    }

    /* ==========================================================================
       RESIZE & PROJECTION MATRIX
       ========================================================================== */
    function resizeCanvas() {
      const container = document.getElementById('canvasWrapper');
      const dpr = window.devicePixelRatio || 1;
      const rect = container.getBoundingClientRect();

      canvas.width = rect.width * dpr;
      canvas.height = rect.height * dpr;
      ctx.resetTransform();
      ctx.scale(dpr, dpr);

      canvas.logicalWidth = rect.width;
      canvas.logicalHeight = rect.height;
    }

    window.addEventListener('resize', resizeCanvas);

    // 3D Perspective Projection Helpers
    // z is distance ahead: 0 is camera/player, 1000 is horizon
    function project3D(trackX, z) {
      const horizonY = canvas.logicalHeight * 0.32;
      const bottomY = canvas.logicalHeight * 0.88;
      const scale = 380 / (z + 380);

      const y = horizonY + (bottomY - horizonY) * scale;
      const trackWidthAtZ = (canvas.logicalWidth * 0.85) * scale;
      const centerX = canvas.logicalWidth * 0.5;
      const x = centerX + (trackX * (trackWidthAtZ * 0.5));

      return { x, y, scale };
    }

    /* ==========================================================================
       LEVEL GENERATION

    function generateLevel(lvl) {
      pickups = [];
      obstacles = [];
      gates = [];
      courseDistance = 0;
      isFinished = false;
      feederMascot.active = false;
      feederMascot.stepMultiplier = 1.0;
      feederMascot.eatenCount = 0;

      // Initial Plate Setup with 3 warm starter pancakes
      player.x = 0;
      player.targetX = 0;
      player.stack = [
        { type: 'PANCAKE', offset: 0 },
        { type: 'PANCAKE', offset: 0 },
        { type: 'PANCAKE', offset: 0 }
      ];
      updateHUD();

      const ingredientKeys = Object.keys(INGREDIENTS);

      // Generate items along track
      for (let z = 350; z < courseLength - 300; z += 120 + Math.random() * 40) {
        const randX = (Math.random() - 0.5) * 1.5;

        // 60% Pickups, 25% Obstacles, 15% Gates
        const roll = Math.random();

        if (roll < 0.62) {
          // Food Pickup
          const chosenKey = ingredientKeys[Math.floor(Math.random() * ingredientKeys.length)];
          pickups.push({
            x: randX,
            z: z,
            type: chosenKey,
            collected: false
          });
        } else if (roll < 0.85) {
          // Rotating Buzzsaw Obstacle
          obstacles.push({
            x: (Math.random() > 0.5 ? 0.5 : -0.5),
            z: z,
            radius: 36,
            angle: 0,
            bladeSpeed: 12 + lvl * 1.5
          });
        } else {
          // Math / Multiplier Gate
          const isMult = Math.random() > 0.5;
          const val = isMult ? 2 : Math.floor(Math.random() * 3 + 2);
          gates.push({
            x: (Math.random() > 0.5 ? 0.45 : -0.45),
            z: z,
            type: isMult ? 'MULTIPLY' : 'ADD',
            value: val,
            passed: false
          });
        }
      }
    }

    /* ==========================================================================
       INPUT LISTENERS (Touch & Mouse Drag)
       ========================================================================== */
    let isDragging = false;
    let lastClientX = 0;

    function handleSteer(deltaX) {
      if (!isPlaying || isFinished) return;
      const sensitivity = 0.0055;
      player.targetX = Math.max(-0.95, Math.min(0.95, player.targetX + deltaX * sensitivity));

      // Hide swipe hint once moved
      const hint = document.getElementById('swipeHint');
      if (hint) hint.style.opacity = '0';
    }

    window.addEventListener('mousedown', (e) => {
      audio.init();
      audio.resume();
      isDragging = true;
      lastClientX = e.clientX;
    });

    window.addEventListener('mousemove', (e) => {
      if (!isDragging) return;
      const dx = e.clientX - lastClientX;
      lastClientX = e.clientX;
      handleSteer(dx);
    });

    window.addEventListener('mouseup', () => { isDragging = false; });

    window.addEventListener('touchstart', (e) => {
      audio.init();
      audio.resume();
      if (e.touches.length > 0) {
        isDragging = true;
        lastClientX = e.touches[0].clientX;
      }
    }, { passive: false });

    window.addEventListener('touchmove', (e) => {
      if (!isDragging || e.touches.length === 0) return;
      const dx = e.touches[0].clientX - lastClientX;
      lastClientX = e.touches[0].clientX;
      handleSteer(dx);
    }, { passive: false });

    window.addEventListener('touchend', () => { isDragging = false; });

    // Keyboard Arrow Steering
    window.addEventListener('keydown', (e) => {
      if (!isPlaying || isFinished) return;
      audio.init();
      audio.resume();
      if (e.key === 'ArrowLeft' || e.key === 'a') player.targetX = Math.max(-0.95, player.targetX - 0.18);
      if (e.key === 'ArrowRight' || e.key === 'd') player.targetX = Math.min(0.95, player.targetX + 0.18);
    });

    function spawnParticles(x, y, color, count = 12) {
      for (let i = 0; i < count; i++) {
        const angle = Math.random() * Math.PI * 2;
        const speed = Math.random() * 5 + 1.5;
        particles.push({
          x,
          y,
          vx: Math.cos(angle) * speed,
          vy: Math.sin(angle) * speed - 1.5,
          color,
          radius: Math.random() * 4 + 2,
          alpha: 1,
          decay: Math.random() * 0.03 + 0.02
        });
      }
    }

    /* ==========================================================================
       PHYSICS & GAMEPLAY UPDATE LOOP
       ========================================================================== */
    let lastTime = 0;

    function update(dt) {
      // Smooth lateral player movement with squash/stretch
      const prevX = player.x;
      player.x += (player.targetX - player.x) * (14 * dt);
      const velocityX = (player.x - prevX) / dt;

      // Elastic wobble angles
      player.wobbleAngle = -velocityX * 0.08;
      player.squash += (1 - player.squash) * (10 * dt);
      player.stretch += (1 - player.stretch) * (10 * dt);

      // Normal Track Movement
      if (!isFinished) {
        courseDistance += scrollSpeed * dt;
        updateHUD();

        // 1. Check Pickups
        pickups.forEach(p => {
          if (p.collected) return;
          const relativeZ = p.z - courseDistance;

          // Pickup in player reach
          if (relativeZ < 40 && relativeZ > -25) {
            const dist = Math.abs(player.x - p.x);
            if (dist < 0.35) {
              p.collected = true;
              const ing = INGREDIENTS[p.type];

              // Stack addition
              player.stack.push({
                type: p.type,
                offset: (Math.random() - 0.5) * 4
              });

              score += ing.points;
              audio.playSquish(ing.pitch);
              player.squash = 1.35;
              player.stretch = 0.75;

              const screenProj = project3D(player.x, 0);
              spawnParticles(screenProj.x, screenProj.y - player.stack.length * 6, ing.color, 14);
              addScoreFlyer(`+${ing.points}`, screenProj.x, screenProj.y - player.stack.length * 6 - 20, ing.color);
              updateHUD();
            }
          }
        });

        // 2. Check Obstacles (Saws)
        obstacles.forEach(o => {
          o.angle += o.bladeSpeed * dt;
          const relativeZ = o.z - courseDistance;

          if (relativeZ < 30 && relativeZ > -20) {
            const dist = Math.abs(player.x - o.x);
            if (dist < 0.38) {
              // Chop off layers!
              if (player.stack.length > 0) {
                const chopCount = Math.min(player.stack.length, 3);
                for (let c = 0; c < chopCount; c++) {
                  player.stack.pop();
                }

                audio.playSlice();
                player.squash = 0.7;
                player.stretch = 1.4;

                const screenProj = project3D(player.x, 0);
                spawnParticles(screenProj.x, screenProj.y, '#f43f5e', 20);
                addScoreFlyer("-CHOPPED!-", screenProj.x, screenProj.y - 40, '#f43f5e');
                updateHUD();

                if (player.stack.length === 0) {
                  triggerGameOver();
                }
              }
            }
          }
        });

        // 3. Check Multiplier Gates
        gates.forEach(g => {
          if (g.passed) return;
          const relativeZ = g.z - courseDistance;

          if (relativeZ < 35 && relativeZ > -20) {
            const dist = Math.abs(player.x - g.x);
            if (dist < 0.45) {
              g.passed = true;
              audio.playChime(784);

              let added = 0;
              if (g.type === 'MULTIPLY') {
                added = player.stack.length * (g.value - 1);
              } else {
                added = g.value;
              }

              for (let i = 0; i < added; i++) {
                player.stack.push({
                  type: 'PANCAKE',
                  offset: (Math.random() - 0.5) * 5
                });
              }

              const screenProj = project3D(player.x, 0);
              spawnParticles(screenProj.x, screenProj.y, '#38bdf8', 25);
              addScoreFlyer(`+${added} PANCAKES!`, screenProj.x, screenProj.y - 60, '#38bdf8');
              updateHUD();
            }
          }
        });

        // Finish Line Trigger
        if (courseDistance >= courseLength) {
          triggerFinishRamp();
        }
      } else {
        // Feeder Mascot Sequence
        handleFinishSequence(dt);
      }

      // Update Particles
      for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.alpha -= p.decay;
        if (p.alpha <= 0) particles.splice(i, 1);
      }

      // Update Floating Text
      for (let i = floatingTexts.length - 1; i >= 0; i--) {
        const ft = floatingTexts[i];
        ft.y += ft.vy;
        ft.alpha -= dt * 1.5;
        if (ft.alpha <= 0) floatingTexts.splice(i, 1);
      }
    }

    /* ==========================================================================
       FINISH & FEAST SEQUENCE (Munching Mascot Bonus Multiplier)
       ========================================================================== */
    function triggerFinishRamp() {
      isFinished = true;
      feederMascot.active = true;
      feederMascot.feedTimer = 0;
      feederMascot.stepMultiplier = 1.0;
      feederMascot.mouthOpen = 1;
    }

    function handleFinishSequence(dt) {
      feederMascot.feedTimer += dt;

      // Pop layers one-by-one into the mascot mouth with crisp ASMR munches
      if (feederMascot.feedTimer > 0.14 && player.stack.length > 0) {
        feederMascot.feedTimer = 0;
        player.stack.pop();
        feederMascot.eatenCount++;
        feederMascot.stepMultiplier = +(feederMascot.stepMultiplier + 0.1).toFixed(1);

        audio.playMunch();
        audio.playChime(440 + feederMascot.eatenCount * 25);

        const screenProj = project3D(0, 180);
        spawnParticles(screenProj.x, screenProj.y - 40, '#f59e0b', 8);

        document.getElementById('multiplierIndicator').textContent = `Feeder Bonus: x${feederMascot.stepMultiplier.toFixed(1)}`;
        updateHUD();
      }

      // All served! Show Level Complete modal
      if (player.stack.length === 0 && !document.getElementById('finishModal').classList.contains('flex')) {
        setTimeout(() => {
          showLevelCompleteModal();
        }, 500);
      }
    }

    function showLevelCompleteModal() {
      isPlaying = false;
      const earned = Math.round(score * feederMascot.stepMultiplier);

      document.getElementById('finalStackVal').textContent = feederMascot.eatenCount.toString();
      document.getElementById('finalMultiplierVal').textContent = `x${feederMascot.stepMultiplier.toFixed(1)}`;
      document.getElementById('finalEarnedVal').textContent = `+${earned}`;

      document.getElementById('finishModal').classList.remove('hidden');
      document.getElementById('finishModal').classList.add('flex');
    }

    function triggerGameOver() {
      isPlaying = false;
      document.getElementById('gameOverModal').classList.remove('hidden');
      document.getElementById('gameOverModal').classList.add('flex');
    }

    /* ==========================================================================
       3D RENDER ENGINE (Stylized Conveyor, Stack Elasticity, & Mascot)
       ========================================================================== */
    function render() {
      const w = canvas.logicalWidth;
      const h = canvas.logicalHeight;

      ctx.clearRect(0, 0, w, h);

      // 1. Draw Stylized Horizon / Sky
      const skyGrad = ctx.createLinearGradient(0, 0, 0, h * 0.4);
      skyGrad.addColorStop(0, '#1c1033');
      skyGrad.addColorStop(1, '#3b1c6e');
      ctx.fillStyle = skyGrad;
      ctx.fillRect(0, 0, w, h);

      // 2. Draw Moving 3D Conveyor Track
      drawConveyorTrack();

      // 3. Draw Distance Gate & Obstacle Objects
      drawTrackEntities();

      // 4. Draw Mascot at Finish (if close)
      const distToFinish = courseLength - courseDistance;
      if (distToFinish < 900) {
        drawFeederMascot(distToFinish);
      }

      // 5. Draw Player Plate & Elastic Wobbly Stack
      drawPlayerStack();

      // 6. Draw Sparkles & Particles
      particles.forEach(p => {
        ctx.save();
        ctx.globalAlpha = p.alpha;
        ctx.fillStyle = p.color;
        ctx.shadowColor = p.color;
        ctx.shadowBlur = 8;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
      });

      // 7. Draw Floating Numbers
      floatingTexts.forEach(ft => {
        ctx.save();
        ctx.globalAlpha = Math.max(0, ft.alpha);
        ctx.font = '900 16px Fredoka';
        ctx.fillStyle = ft.color;
        ctx.shadowColor = '#000000';
        ctx.shadowBlur = 6;
        ctx.textAlign = 'center';
        ctx.fillText(ft.text, ft.x, ft.y);
        ctx.restore();
      });
    }

    function drawConveyorTrack() {
      // Perspective polygon for the track
      const pFarLeft = project3D(-1.1, 950);
      const pFarRight = project3D(1.1, 950);
      const pNearRight = project3D(1.1, 0);
      const pNearLeft = project3D(-1.1, 0);

      // Track Surface Gradient
      const trackGrad = ctx.createLinearGradient(0, pFarLeft.y, 0, pNearLeft.y);
      trackGrad.addColorStop(0, '#431407');
      trackGrad.addColorStop(0.5, '#78350f');
      trackGrad.addColorStop(1, '#92400e');

      ctx.beginPath();
      ctx.moveTo(pFarLeft.x, pFarLeft.y);
      ctx.lineTo(pFarRight.x, pFarRight.y);
      ctx.lineTo(pNearRight.x, pNearRight.y);
      ctx.lineTo(pNearLeft.x, pNearLeft.y);
      ctx.closePath();
      ctx.fillStyle = trackGrad;
      ctx.fill();

      // Track Glowing Curbs
      ctx.strokeStyle = '#f59e0b';
      ctx.lineWidth = 3;
      ctx.stroke();

      // Conveyor Chevron Markings
      const chevronStep = 80;
      const offset = (courseDistance % chevronStep);

      for (let z = -offset; z < 950; z += chevronStep) {
        if (z < 0) continue;
        const leftP = project3D(-0.9, z);
        const centerP = project3D(0, z + 25);
        const rightP = project3D(0.9, z);

        ctx.strokeStyle = 'rgba(254, 243, 199, 0.25)';
        ctx.lineWidth = 2.5 * leftP.scale;
        ctx.beginPath();
        ctx.moveTo(leftP.x, leftP.y);
        ctx.lineTo(centerP.x, centerP.y);
        ctx.lineTo(rightP.x, rightP.y);
        ctx.stroke();
      }
    }

    function drawTrackEntities() {
      // Sort items by distance from far to near for correct rendering order
      const entities = [];

      pickups.forEach(p => {
        const relZ = p.z - courseDistance;
        if (relZ > -50 && relZ < 950 && !p.collected) {
          entities.push({ type: 'PICKUP', data: p, z: relZ });
        }
      });

      obstacles.forEach(o => {
        const relZ = o.z - courseDistance;
        if (relZ > -50 && relZ < 950) {
          entities.push({ type: 'OBSTACLE', data: o, z: relZ });
        }
      });

      gates.forEach(g => {
        const relZ = g.z - courseDistance;
        if (relZ > -50 && relZ < 950 && !g.passed) {
          entities.push({ type: 'GATE', data: g, z: relZ });
        }
      });

      entities.sort((a, b) => b.z - a.z);

      entities.forEach(ent => {
        if (ent.type === 'PICKUP') drawPickupItem(ent.data, ent.z);
        if (ent.type === 'OBSTACLE') drawBuzzsaw(ent.data, ent.z);
        if (ent.type === 'GATE') drawMultiplierGate(ent.data, ent.z);
      });
    }

    function drawPickupItem(pickup, relZ) {
      const proj = project3D(pickup.x, relZ);
      const ing = INGREDIENTS[pickup.type];
      const r = ing.radius * proj.scale;
      const h = ing.height * proj.scale;

      ctx.save();
      ctx.translate(proj.x, proj.y);

      // Tactile shadow
      ctx.beginPath();
      ctx.ellipse(0, 4, r * 1.05, r * 0.45, 0, 0, Math.PI * 2);
      ctx.fillStyle = 'rgba(0, 0, 0, 0.35)';
      ctx.fill();

      // Cylinder 3D body
      ctx.fillStyle = ing.rimColor;
      ctx.beginPath();
      ctx.ellipse(0, 0, r, r * 0.45, 0, 0, Math.PI * 2);
      ctx.ellipse(0, -h, r, r * 0.45, 0, 0, Math.PI * 2);
      ctx.fill();

      ctx.fillRect(-r, -h, r * 2, h);

      // Glossy Top Cap
      const topGrad = ctx.createRadialGradient(-r * 0.3, -h - (r * 0.2), 1, 0, -h, r);
      topGrad.addColorStop(0, '#ffffff');
      topGrad.addColorStop(0.3, ing.color);
      topGrad.addColorStop(1, ing.rimColor);

      ctx.beginPath();
      ctx.ellipse(0, -h, r, r * 0.45, 0, 0, Math.PI * 2);
      ctx.fillStyle = topGrad;
      ctx.fill();
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.4)';
      ctx.lineWidth = 1.5;
      ctx.stroke();

      ctx.restore();
    }

    function drawBuzzsaw(saw, relZ) {
      const proj = project3D(saw.x, relZ);
      const r = saw.radius * proj.scale;

      ctx.save();
      ctx.translate(proj.x, proj.y - 10 * proj.scale);

      // Rotating saw blade
      ctx.rotate(saw.angle);
      ctx.fillStyle = '#e2e8f0';
      ctx.strokeStyle = '#475569';
      ctx.lineWidth = 2;

      // Draw 8 jagged teeth
      ctx.beginPath();
      const teeth = 8;
      for (let i = 0; i < teeth; i++) {
        const a1 = (i / teeth) * Math.PI * 2;
        const a2 = ((i + 0.5) / teeth) * Math.PI * 2;
        ctx.lineTo(Math.cos(a1) * r, Math.sin(a1) * r);
        ctx.lineTo(Math.cos(a2) * (r * 0.65), Math.sin(a2) * (r * 0.65));
      }
      ctx.closePath();
      ctx.fill();
      ctx.stroke();

      // Center red bolt
      ctx.beginPath();
      ctx.arc(0, 0, r * 0.28, 0, Math.PI * 2);
      ctx.fillStyle = '#ef4444';
      ctx.fill();

      ctx.restore();
    }

    function drawMultiplierGate(gate, relZ) {
      const proj = project3D(gate.x, relZ);
      const gw = 110 * proj.scale;
      const gh = 130 * proj.scale;

      ctx.save();
      ctx.translate(proj.x, proj.y - gh * 0.6);

      // Glowing Archway
      const isMult = gate.type === 'MULTIPLY';
      ctx.fillStyle = isMult ? 'rgba(56, 189, 248, 0.55)' : 'rgba(74, 222, 128, 0.55)';
      ctx.strokeStyle = isMult ? '#38bdf8' : '#4ade80';
      ctx.lineWidth = 3;

      ctx.beginPath();
      ctx.roundRect(-gw / 2, -gh / 2, gw, gh, 12 * proj.scale);
      ctx.fill();
      ctx.stroke();

      // Gate Text Label
      ctx.font = `900 ${Math.round(26 * proj.scale)}px Fredoka`;
      ctx.fillStyle = '#ffffff';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.shadowColor = '#000000';
      ctx.shadowBlur = 6;
      ctx.fillText(isMult ? `x${gate.value}` : `+${gate.value}`, 0, 0);

      ctx.restore();
    }

    function drawFeederMascot(distAhead) {
      const proj = project3D(0, Math.max(120, distAhead));
      const s = proj.scale * 1.5;

      ctx.save();
      ctx.translate(proj.x, proj.y - 60 * s);

      // Giant Hungry Cute Chef Mascot
      // Head / Body
      ctx.fillStyle = '#fed7aa';
      ctx.beginPath();
      ctx.arc(0, 0, 55 * s, 0, Math.PI * 2);
      ctx.fill();

      // Rosy Cheeks
      ctx.fillStyle = 'rgba(244, 63, 94, 0.4)';
      ctx.beginPath();
      ctx.arc(-32 * s, 10 * s, 12 * s, 0, Math.PI * 2);
      ctx.arc(32 * s, 10 * s, 12 * s, 0, Math.PI * 2);
      ctx.fill();

      // Big Kawaii Eyes
      ctx.fillStyle = '#0f172a';
      ctx.beginPath();
      ctx.arc(-22 * s, -12 * s, 9 * s, 0, Math.PI * 2);
      ctx.arc(22 * s, -12 * s, 9 * s, 0, Math.PI * 2);
      ctx.fill();

      // Eye White Sparkles
      ctx.fillStyle = '#ffffff';
      ctx.beginPath();
      ctx.arc(-20 * s, -14 * s, 3.5 * s, 0, Math.PI * 2);
      ctx.arc(24 * s, -14 * s, 3.5 * s, 0, Math.PI * 2);
      ctx.fill();

      // Giant Open Mouth (Feeder Chasm)
      ctx.fillStyle = '#881337';
      ctx.beginPath();
      ctx.ellipse(0, 22 * s, 30 * s, (feederMascot.mouthOpen ? 28 : 18) * s, 0, 0, Math.PI * 2);
      ctx.fill();

      // White Chef Hat
      ctx.fillStyle = '#ffffff';
      ctx.beginPath();
      ctx.roundRect(-40 * s, -85 * s, 80 * s, 40 * s, 16 * s);
      ctx.fill();

      ctx.restore();
    }

    function drawPlayerStack() {
      const proj = project3D(player.x, 0);

      ctx.save();
      ctx.translate(proj.x, proj.y);

      // Plate Base (Glossy Ceramic)
      ctx.fillStyle = 'rgba(0, 0, 0, 0.4)';
      ctx.beginPath();
      ctx.ellipse(0, 6, 55, 24, 0, 0, Math.PI * 2);
      ctx.fill();

      const plateGrad = ctx.createLinearGradient(0, -10, 0, 10);
      plateGrad.addColorStop(0, '#ffffff');
      plateGrad.addColorStop(0.5, '#e2e8f0');
      plateGrad.addColorStop(1, '#94a3b8');

      ctx.fillStyle = plateGrad;
      ctx.beginPath();
      ctx.ellipse(0, 0, 52, 22, 0, 0, Math.PI * 2);
      ctx.fill();
      ctx.strokeStyle = '#cbd5e1';
      ctx.lineWidth = 2;
      ctx.stroke();

      // Draw Stacking Layers with Squash, Stretch & Progressive Elastic Wobble
      let currentHeightOffset = 4;

      player.stack.forEach((item, index) => {
        const ing = INGREDIENTS[item.type];
        const wobbleFactor = (index + 1) * player.wobbleAngle * 4;

        ctx.save();
        ctx.translate(item.offset + wobbleFactor, -currentHeightOffset);
        ctx.scale(player.squash, player.stretch);

        const r = ing.radius;
        const h = ing.height;

        // Rim
        ctx.fillStyle = ing.rimColor;
        ctx.beginPath();
        ctx.ellipse(0, 0, r, r * 0.44, 0, 0, Math.PI * 2);
        ctx.ellipse(0, -h, r, r * 0.44, 0, 0, Math.PI * 2);
        ctx.fill();
        ctx.fillRect(-r, -h, r * 2, h);

        // Cap
        const capGrad = ctx.createRadialGradient(-r * 0.3, -h - 4, 1, 0, -h, r);
        capGrad.addColorStop(0, '#ffffff');
        capGrad.addColorStop(0.35, ing.color);
        capGrad.addColorStop(1, ing.rimColor);

        ctx.beginPath();
        ctx.ellipse(0, -h, r, r * 0.44, 0, 0, Math.PI * 2);
        ctx.fillStyle = capGrad;
        ctx.fill();
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.35)';
        ctx.lineWidth = 1;
        ctx.stroke();

        ctx.restore();
        currentHeightOffset += (ing.height + 2);
      });

      ctx.restore();
    }

    /* ==========================================================================
       GAME ENGINE TICK
       ========================================================================== */
    function gameLoop(timestamp) {
      if (!lastTime) lastTime = timestamp;
      const dt = Math.min((timestamp - lastTime) / 1000, 0.1);
      lastTime = timestamp;

      if (isPlaying) {
        update(dt);
      }
      render();

      requestAnimationFrame(gameLoop);
    }
